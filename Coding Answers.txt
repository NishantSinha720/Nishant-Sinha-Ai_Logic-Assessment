def find_min(nums):
    left, right = 0, len(nums) - 1
    
    while left < right:
       mid = (left + right) // 2
	
       if nums[mid] > nums[right]:
       

          left = mid + 1
       else:

          right = mid

    return nums[left]

#Example 
print(find_min([3,4,5,1,2]))    #1

print(find_min([4,5,6,7,0,1,2])) #2

print(find_min([1,2,3]))         #1

print(find_min([5]))             #5

#Complexity Time: O(log n); extra space 0(1)

def length_of_longest_substring(s):
    last_seen = {}
    left = 0
    best = 0


    for right, char in enumerate(s):
        if char in last_seen:
           left = max(left, last_seen[char] + 1)
  
        last_seen[char] = right
        best = max(best, right - left + 1
     
    return best


#Example

print(length_of_substring("abcabcbb")) #3

print(length_of_substring("bbbbb"))    #1

print(length_of_substring("ppwwkew"))  #3

print(length_of_substring("abba"))     #2

print(length_of_substring(""))         #0


#Complexity Time: 0(n)

def next_permutation(nums):

    i = len(nums) -2
    

    while i >=0 and nums[i] >=nums[i +1}:

       i -= 1


    if i >= 0:

       j = len(nums) - 1  


       while nums[j] <= nums[i]:
          
           j -= 1



       nums[1], nums[j] = nums[j], nums[i]


      left, right = i + 1, len(nums) - 1


      while left < right:
          nums[left], nums[right] = nums[right], nums[left]
          left += 1
          right -= 1


Examples:

nums = [1,2,3] next_permutation(nums) print(nums)    #[1,3,2]

nums = [3,2,1] next_permutation(nums) print(nums)    #[1,2,3]

nums = [1,1,5] next_permutation(nums) print(nums)    #[1,5,1]

Complexity Time = O(n): extra space 0(1)
